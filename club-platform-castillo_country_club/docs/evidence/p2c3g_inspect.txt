   129 /**
   130  * Execute an MCP agent tool as a mutation (write operation).
   131  *
   132  * Automatically invalidates all queries for the same agent
   133  * on success, triggering background refetch of stale data.
   134  *
   135  * Spread order is CRITICAL:
   136  *   1. ...options first (onError, onMutate, onSettled pass through)
   137  *   2. mutationFn override (always routes through callMCPTool)
   138  *   3. onSuccess override (always invalidates + delegates to user callback)
   139  *
   140  * This ensures user callbacks are NEVER silently replaced,
   141  * and cache invalidation NEVER depends on whether the user passes onSuccess.
   142  *
   143  * @typeParam T — Expected response type
   144  * @typeParam P — Parameters type (must extend Record<string, unknown>)
   145  * @param agent — MCP server/agent name
   146  * @param tool — Tool name
   147  * @param options — React Query mutation options (onSuccess, onError, etc.)
   148  * @returns React Query mutation result with mutate, isLoading, error, etc.
   149  *
   150  * @example
   151  * const mutation = useMCPMutation<CreateEventResponse, CreateEventParams>(
   152  *   "events", "create_event",
   153  *   {
   154  *     onSuccess: (data) => {
   155  *       showToast({ type: "success", message: "Event created" });
   156  *     },
   157  *   }
   158  * );
   159  *
   160  * mutation.mutate({ title: "Gala", date: "2026-09-01" });
   161  */
   162 export function useMCPMutation<
   163   T = unknown,
   164   P extends Record<string, unknown> = Record<string, unknown>,
   165 >(agent: string, tool: string, options?: UseMutationOptions<T, Error, P>) {
   166   const queryClient = useQueryClient();
   167 
   168   return useMutation<T, Error, P>({
   169     // Step 1: Spread user options FIRST
   170     // onError, onMutate, onSettled, retry, etc. pass through unchanged
   171     ...options,
   172 
   173     // Step 2: Override mutationFn — ALWAYS route through callMCPTool
   174     // Users should not replace the transport layer
   175     mutationFn: (params: P) => callMCPTool<T>(agent, tool, params),
   176 
   177     // Step 3: Override onSuccess — ALWAYS invalidate + delegate
   178     // React Query v5 signature: (data, variables, context) — 3 params
   179     // context is the return value of onMutate (if provided)
   180     onSuccess: (data, variables, onMutateResult, context) => {
   181       // Invalidate all queries for this agent (fires refetch)
   182       queryClient.invalidateQueries({ queryKey: [agent] });
   183 
   184       // Delegate to user's onSuccess if provided
   185       // Returning the result preserves Promise chaining for async callbacks
   186       return options?.onSuccess?.(data, variables, onMutateResult, context);
   187     },
   188   });
   189 }
