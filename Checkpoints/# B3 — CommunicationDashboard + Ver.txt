## Archivo único — `CommunicationDashboard.tsx` (NUEVO — completo)

```tsx
// src/features/communications/CommunicationDashboard.tsx
// ═══════════════════════════════════════════════════════════
// Communication Intelligence Dashboard — Castillo Country Club
// Phase 11 — The Hub
//
// Integrates:
//   Pulso        → CommunicationOverview (NEW)
//   Centro       → CommunicationCenter (EXISTING)
//   Tablón       → AnnouncementBoard (EXISTING)
//   Analytics    → CommunicationAnalytics (NEW)
//   Herramientas → ScheduleComposer + CommunicationPreferences (NEW)
//
// Rules: #13 DetailTabs, #16 breadcrumbs, verification
// ═══════════════════════════════════════════════════════════

import { useState, useCallback, lazy, Suspense } from "react";
import {
  Heart, Send, Megaphone, BarChart3, Wrench,
  CalendarClock, UserCog,
} from "lucide-react";
import { PageContainer } from "@/shared/components/layout/PageContainer";
import { DetailTabs, type DetailTab } from "@/shared/components/layout/DetailTabs";
import { Skeleton } from "@/shared/components/ui";
import { cn } from "@/shared/lib/utils";

/* ══════════════════════════════════════════════════════════
   Section 1: Lazy-loaded Tab Panels
   ══════════════════════════════════════════════════════════ */

// NEW components (Phase 11)
const CommunicationOverview = lazy(() => import("./components/CommunicationOverview"));
const CommunicationAnalytics = lazy(() => import("./components/CommunicationAnalytics"));
const ScheduleComposer = lazy(() => import("./components/ScheduleComposer"));
const CommunicationPreferences = lazy(() => import("./components/CommunicationPreferences"));

// EXISTING components (Phase 10 / prior)
const CommunicationCenter = lazy(() => import("./CommunicationCenter"));
const AnnouncementBoard = lazy(() => import("./AnnouncementBoard"));

/* ══════════════════════════════════════════════════════════
   Section 2: Tab Panel Suspense Wrapper
   ══════════════════════════════════════════════════════════ */

function TabPanel({ children }: { children: React.ReactNode }) {
  return (
    <Suspense
      fallback={
        <div className="mt-6 space-y-4">
          <Skeleton className="h-8 w-64 rounded-lg" />
          <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
            {Array.from({ length: 6 }).map((_, i) => (
              <Skeleton key={i} className="h-40 rounded-xl" />
            ))}
          </div>
        </div>
      }
    >
      {children}
    </Suspense>
  );
}

/* ══════════════════════════════════════════════════════════
   Section 3: Herramientas Sub-Tabs
   ══════════════════════════════════════════════════════════ */

type HerramientaTab = "programar" | "preferencias";

function HerramientasPanel() {
  const [subTab, setSubTab] = useState<HerramientaTab>("programar");

  const subTabs: Array<{ id: HerramientaTab; label: string; icon: typeof CalendarClock }> = [
    { id: "programar", label: "Programar Envío", icon: CalendarClock },
    { id: "preferencias", label: "Preferencias Socio", icon: UserCog },
  ];

  return (
    <div className="space-y-6 mt-6">
      {/* Sub-tab bar */}
      <div className="flex items-center gap-1 p-1 rounded-xl bg-[var(--color-surface-1)] border border-white/[0.06] w-fit">
        {subTabs.map((tab) => {
          const isActive = subTab === tab.id;
          const Icon = tab.icon;
          return (
            <button
              key={tab.id}
              onClick={() => setSubTab(tab.id)}
              className={cn(
                "flex items-center gap-2 px-4 py-2.5 rounded-lg text-sm font-medium",
                "font-[family-name:var(--font-ui)] transition-all",
                "focus-visible:outline-2 focus-visible:outline-[var(--color-gold-400)]",
                "min-h-[44px]",
                isActive
                  ? "bg-[var(--color-surface-2)] text-[var(--color-silver-200)] shadow-sm border border-white/[0.08]"
                  : "text-[var(--color-silver-500)] hover:text-[var(--color-silver-300)] border border-transparent",
              )}
              aria-pressed={isActive}
            >
              <Icon className="w-4 h-4" />
              {tab.label}
            </button>
          );
        })}
      </div>

      {/* Sub-tab content */}
      {subTab === "programar" && <ScheduleComposer />}
      {subTab === "preferencias" && <CommunicationPreferences />}
    </div>
  );
}

/* ══════════════════════════════════════════════════════════
   Section 4: Main Component
   ══════════════════════════════════════════════════════════ */

export default function CommunicationDashboard() {
  const [activeTab, setActiveTab] = useState("pulso");

  /* ── Tab navigation callback for Overview quick actions ── */
  const handleNavigate = useCallback((tab: string) => {
    setActiveTab(tab);
  }, []);

  /* ── Tab definitions ── */
  const tabs: DetailTab[] = [
    { id: "pulso", label: "Pulso", icon: Heart },
    { id: "centro", label: "Centro", icon: Send },
    { id: "tablon", label: "Tablón", icon: Megaphone },
    { id: "analytics", label: "Analytics", icon: BarChart3 },
    { id: "herramientas", label: "Herramientas", icon: Wrench },
  ];

  return (
    <PageContainer
      title="Communication Intelligence"
      subtitle="Plataforma centralizada de inteligencia comunicacional"
      breadcrumbs={[
        { label: "Dashboard", href: "/admin" },
        { label: "Comunicaciones", href: "/admin/communications" },
      ]}
    >
      <div className="space-y-6">
        {/* ── Tab Navigation ── */}
        <DetailTabs
          tabs={tabs}
          active={activeTab}
          onChange={setActiveTab}
          ariaLabel="Secciones de Communication Intelligence"
        />

        {/* ── Tab Panels ── */}
        {activeTab === "pulso" && (
          <TabPanel>
            <CommunicationOverview onNavigate={handleNavigate} />
          </TabPanel>
        )}

        {activeTab === "centro" && (
          <TabPanel>
            <CommunicationCenter />
          </TabPanel>
        )}

        {activeTab === "tablon" && (
          <TabPanel>
            <AnnouncementBoard />
          </TabPanel>
        )}

        {activeTab === "analytics" && (
          <TabPanel>
            <CommunicationAnalytics />
          </TabPanel>
        )}

        {activeTab === "herramientas" && (
          <TabPanel>
            <HerramientasPanel />
          </TabPanel>
        )}
      </div>
    </PageContainer>
  );
}
```

### Decisiones de diseño

| # | Decisión | Razón |
|---|---------|-------|
| 1 | `lazy()` + `Suspense` para todos los paneles | Performance: solo carga el componente del tab activo. CommunicationCenter + AnnouncementBoard son pesados |
| 2 | Herramientas con sub-tab bar (no DetailTabs) | Patrón Fase 10: solo 2 items → toggle simple. DetailTabs es para 3+ con ARIA |
| 3 | `onNavigate` callback desde Overview | Desacoplamiento total — Overview navega al Centro o Herramientas sin conocer la estructura del dashboard |
| 4 | EmergencyBroadcast excluido | Confirmado: es standalone, accesible desde CommunicationCenter (EmergencyModal) |
| 5 | PageContainer con breadcrumbs absolutos | Regla #16: `/admin` → `/admin/communications` |
