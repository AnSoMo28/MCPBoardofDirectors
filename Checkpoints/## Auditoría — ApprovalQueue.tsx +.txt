## Auditoría — ApprovalQueue.tsx + types.ts

Este componente es **el mejor escrito de los 4**. La estructura es sólida. Los hallazgos son menores comparados con los anteriores.

### Problemas

| # | Problema | Severidad |
|---|----------|----------|
| 1 | **Agent name mismatch**: `"communication_agent"` en AGENT_LABELS y AGENT_ACCENT → backend usa `"communication"` (CommunicationAgent.agent_name = "communication") | ALTA — labels y colores no matchean para approvals de comunicación |
| 2 | **Motion variants llamadas en JSX**: `useMotionVariants()` se llama 2 veces dentro del render en vez de una vez al top del componente | MEDIA — performance |
| 3 | **Sin stats summary**: no hay overview de cuántos por nivel de aprobación (HITL vs ALWAYS_HUMAN), ni por agente | MEDIA — UX |
| 4 | **Tab bar local** para status filter en vez de DetailTabs (Regla #13) | MEDIA |
| 5 | **Sin `useBreakpoint()`** | MEDIA — Regla #20 |
| 6 | **Rechazo sin nota obligatoria**: se puede rechazar sin explicar por qué | MEDIA — operacional |
| 7 | **Sin `enabled` en query**: el tool `get_pending_approvals` se llama siempre, aunque estés en otra página | BAJA |
| 8 | **No muestra las 17 tipologías del CLAUDE_CONTEXT**: el approval card genérico no distingue visualmente entre "finance_expense_create" (alto riesgo) y "wellness_schedule_class" (bajo riesgo) | BAJA |
| 9 | **`focus-visible:border-`** sin `outline-` explícito en textarea | BAJA |

/** Risk keywords for coloring risk_summary */
function getRiskColor(text: string | null | undefined): string {
  if (!text) return "var(--color-silver-400)";
  const lower = text.toLowerCase();
  if (lower.includes("crítico") || lower.includes("critical") || lower.includes("alto") || lower.includes("emergency"))
    return "var(--color-rose-400)";
  if (lower.includes("medio") || lower.includes("medium") || lower.includes("masivo"))
    return "var(--color-amber-400)";
  return "var(--color-silver-400)";
}

function ApprovalStats({
  requests,
}: {
  requests: ApprovalRequest[];
}) {
  const stats = useMemo(() => {
    const pending = requests.filter((r) => r.status === "pending");
    const dual = pending.filter((r) => r.approval_level === "always_human");
    const byAgent: Record<string, number> = {};
    for (const r of pending) {
      byAgent[r.agent] = (byAgent[r.agent] || 0) + 1;
    }
    const topAgent = Object.entries(byAgent).sort(([, a], [, b]) => b - a)[0];
    return {
      total: requests.length,
      pending: pending.length,
      dual: dual.length,
      approved: requests.filter((r) => r.status === "approved").length,
      rejected: requests.filter((r) => r.status === "rejected").length,
      topAgent: topAgent ? { name: AGENT_LABELS[topAgent[0]] ?? topAgent[0], count: topAgent[1] } : null,
    };
  }, [requests]);

  if (stats.total === 0) return null;

  const items = [
    { label: "Pendientes", value: stats.pending.toString(), color: "var(--color-amber-400)", icon: Clock },
    { label: "Doble Aprob.", value: stats.dual.toString(), color: "var(--color-rose-400)", icon: ShieldAlert },
    { label: "Aprobados", value: stats.approved.toString(), color: "var(--color-emerald-400)", icon: CheckCircle2 },
    { label: "Rechazados", value: stats.rejected.toString(), color: "var(--color-rose-400)", icon: XCircle },
  ];

  return (
    <div className="grid grid-cols-2 lg:grid-cols-4 gap-3">
      {items.map((item) => {
        const Icon = item.icon;
        return (
          <Card key={item.label} className="p-4 relative overflow-hidden">
            <div
              className="absolute top-0 right-0 w-12 h-12 rounded-full opacity-[0.07] -translate-y-4 translate-x-4"
              style={{ backgroundColor: item.color }}
            />
            <div className="relative flex items-center gap-3">
              <div
                className="p-1.5 rounded-lg flex-shrink-0"
                style={{
                  backgroundColor: `color-mix(in srgb, ${item.color} 12%, transparent)`,
                  color: item.color,
                }}
              >
                <Icon className="w-4 h-4" />
              </div>
              <div>
                <p className="text-xs font-[family-name:var(--font-ui)] text-[var(--color-silver-400)]">{item.label}</p>
                <p className="text-xl font-bold font-[family-name:var(--font-display)] text-[var(--color-silver-200)]">
                  {item.value}
                </p>
              </div>
            </div>
          </Card>
        );
      })}
    </div>
  );
}

  const motionV = useMotion();
  const [activeTab, setActiveTab] = useState<ViewTab>("pending");
  const [agentFilter, setAgentFilter] = useState("");
  const [search, setSearch] = useState("");
  const [processingId, setProcessingId] = useState<string | null>(null);

  /* ── Queries ── */
  const { data, isLoading, refetch } = useMCPQuery<ApprovalListResponse>(
    "hitl",
    "get_pending_approvals",
    {},
    { refetchInterval: 30_000 },
  );

  /* ── Mutations ── */
  const approveMut = useMCPMutation<
    unknown,
    { request_id: string; approved_by: string; note: string }
  >("hitl", "approve_request", {
    onSuccess: () => {
      showToast({ type: "success", message: "Solicitud aprobada correctamente" });
      setProcessingId(null);
      refetch();
    },
    onError: (err) => {
      showToast({ type: "error", message: err instanceof Error ? err.message : "Error al aprobar" });
      setProcessingId(null);
    },
  });

  const rejectMut = useMCPMutation<
    unknown,
    { request_id: string; rejected_by: string; reason: string }
  >("hitl", "reject_request", {
    onSuccess: () => {
      showToast({ type: "info", message: "Solicitud rechazada" });
      setProcessingId(null);
      refetch();
    },
    onError: (err) => {
      showToast({ type: "error", message: err instanceof Error ? err.message : "Error al rechazar" });
      setProcessingId(null);
    },
  });

  /* ── Data ── */
  const requests = data?.requests ?? [];
  const pendingCount = requests.filter((r) => r.status === "pending").length;
  const dualCount = requests.filter((r) => r.status === "pending" && r.approval_level === "always_human").length;

  /* ── Tabs ── */
  const tabs: DetailTab[] = [
    { id: "pending", label: "Pendientes", icon: Clock, count: pendingCount },
    { id: "all", label: "Todas", icon: Shield, count: requests.length },
  ];

  /* ── Filtered data ── */
  const filtered = useMemo(() => {
    let list = requests;
    if (activeTab === "pending") list = list.filter((r) => r.status === "pending");
    if (agentFilter) list = list.filter((r) => r.agent === agentFilter);
    if (search) {
      const q = search.toLowerCase();
      list = list.filter(
        (r) =>
          r.tool.toLowerCase().includes(q) ||
          r.risk_summary?.toLowerCase().includes(q) ||
          (AGENT_LABELS[r.agent] ?? r.agent).toLowerCase().includes(q),
      );
    }
    return list;
  }, [requests, activeTab, agentFilter, search]);

  /* ── Active agents (for filter dropdown) ── */
  const activeAgents = useMemo(() => {
    const set = new Set(requests.map((r) => r.agent));
    return Array.from(set).sort();
  }, [requests]);

  /* ── Handlers ── */
  const handleApprove = useCallback(
    (requestId: string, note: string) => {
      setProcessingId(requestId);
      approveMut.mutate({
        request_id: requestId,
        approved_by: user?.name ?? "unknown",
        note,
      });
    },
    [approveMut, user],
  );

  const handleReject = useCallback(
    (requestId: string, reason: string) => {
      setProcessingId(requestId);
      rejectMut.mutate({
        request_id: requestId,
        rejected_by: user?.name ?? "unknown",
        reason,
      });
    },
    [rejectMut, user],
  );

  /* ── Render ── */
  return (
    <PageContainer
      title="Cola de Aprobaciones"
      subtitle={
        pendingCount > 0
          ? `${pendingCount} pendiente${pendingCount !== 1 ? "s" : ""}${dualCount > 0 ? ` · ${dualCount} doble confirmación` : ""}`
          : "Sin solicitudes pendientes"
      }
    >
      <div className="space-y-6">
        {/* Stats */}
        {!isLoading && <ApprovalStats requests={requests} />}

        {/* Tab Navigation */}
        <DetailTabs
          tabs={tabs}
          active={activeTab}
          onChange={(id) => setActiveTab(id as ViewTab)}
          ariaLabel="Filtro de aprobaciones"
        />

        {/* Filters */}
        <div className="flex flex-col sm:flex-row items-start sm:items-center gap-3">
          <div className="relative flex-1 max-w-xs">
            <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-[var(--color-silver-400)]" />
            <input
              type="text"
              value={search}
              onChange={(e) => setSearch(e.target.value)}
              placeholder="Buscar aprobaciones…"
              className="w-full pl-10 pr-4 py-2.5 rounded-lg border text-sm font-[family-name:var(--font-ui)]
                         bg-[var(--color-surface-1)] text-[var(--color-silver-200)] border-white/[0.06]
                         focus-visible:outline-2 focus-visible:outline-[var(--color-gold-400)] min-h-[44px]"
              aria-label="Buscar aprobaciones"
            />
          </div>

          {activeAgents.length > 1 && (
            <select
              value={agentFilter}
              onChange={(e) => setAgentFilter(e.target.value)}
              className="rounded-lg border px-3 py-2 text-sm font-[family-name:var(--font-ui)]
                         bg-[var(--color-surface-1)] text-[var(--color-silver-200)] border-white/[0.06]
                         focus-visible:outline-2 focus-visible:outline-[var(--color-gold-400)] min-h-[44px]"
              aria-label="Filtrar por agente"
            >
              <option value="">Todos los agentes</option>
              {activeAgents.map((a) => (
                <option key={a} value={a}>{AGENT_LABELS[a] ?? a}</option>
              ))}
            </select>
          )}
        </div>

        {/* Content */}
        {isLoading ? (
          <div className="space-y-3">
            {Array.from({ length: 3 }).map((_, i) => (
              <Skeleton key={i} className="h-28 rounded-[var(--radius-lg)]" />
            ))}
          </div>
        ) : filtered.length === 0 ? (
          <EmptyState
            icon={<Shield className="w-12 h-12" />}
            title={
              activeTab === "pending"
                ? "Sin aprobaciones pendientes"
                : "Sin solicitudes de aprobación"
            }
            description={
              activeTab === "pending"
                ? "Todas las solicitudes fueron procesadas."
                : "No hay solicitudes registradas."
            }
          />
        ) : (
          <motion.div
            variants={motionV.list}
            initial={false}
            animate="visible"
            className="space-y-3"
          >
            {filtered.map((request) => (
              <motion.div key={request.request_id} variants={motionV.item}>
                <ApprovalCard
                  request={request}
                  onApprove={handleApprove}
                  onReject={handleReject}
                  isProcessing={processingId === request.request_id}
                />
              </motion.div>
            ))}
          </motion.div>
        )}
      </div>
    </PageContainer>
  );
}
